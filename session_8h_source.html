52#define SOCK_TRY_RECV(fd, buff, type, action) \
53 toread = sizeof(type); \
54 while (1) { \
55 res = recv(fd, buff, toread, 0); \
56 if (res < 0) { \
57 ERROR("recv(): %s", strerror(errno)); \
58 action; break; \
59 } else if (res == 0) { \
60 DEBUG("connection closed by peer"); \
61 action; break; \
62 } else if (res < sizeof(type)) { \

64 toread -= res; \
